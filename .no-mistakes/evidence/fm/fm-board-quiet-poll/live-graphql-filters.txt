--- Scenario E: the adapter's own GraphQL filters resolve a real board and a real card ---
# fields_jq and card_jq are taken verbatim from bin/fm-board.sh and run by gh against live GitHub.
# read-only: no mutation is sent.

$ fields_jq Status     # the filter the adapter generates for status-field = Status
(.data.repositoryOwner.projectV2 // empty)
| (["project", (.id // "" | tostring)] | @tsv),
  ( ((.fields.nodes // [])[]
     | select(((.name // "") | ascii_downcase | gsub(" "; "")) == "status")
     | (["field", (.id // "" | tostring)] | @tsv),
       ((.options // [])[] | ["option", (.id // "" | tostring), (.name // "" | tostring)] | @tsv)
    ) )

$ gh api graphql -f query="$BOARD_IDS_QUERY" -f owner=huang-daniel -F number=2 -F fields=100 --jq "$(fields_jq Status)"

$ card_jq huang-daniel 2
[ ((.data.repository.issue.projectItems.nodes // [])[]
   | select((((.project.owner.login // "") | ascii_downcase | gsub(" "; "")) == "huang-daniel")
            and (((.project.number // "") | tostring) == "2"))
   | (.id // "" | tostring)
   | select(. != "")) ]
| first // empty

$ gh api graphql -f query="$CARD_QUERY" -f owner=huang-daniel -f name=gamba-labs-main -F number=3 -F projects=20 --jq "$(card_jq huang-daniel 2)"
